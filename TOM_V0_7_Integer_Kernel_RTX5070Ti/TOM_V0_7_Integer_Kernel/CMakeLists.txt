cmake_minimum_required(VERSION 3.24)
project(TOMIntegerValidation VERSION 0.7.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(TOM_ENABLE_CUDA "Build the CUDA backend" ON)
set(TOM_CUDA_ARCH "120" CACHE STRING "RTX 5070 Ti Laptop compute target")
add_library(tom_headers INTERFACE)
target_include_directories(tom_headers INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)
add_executable(tom_cpu src/cpu_main.cpp)
target_link_libraries(tom_cpu PRIVATE tom_headers)
add_executable(tom_native_cpu src/native_cpu.cpp)
target_link_libraries(tom_native_cpu PRIVATE tom_headers)
add_executable(tom_core_tests tests/core_tests.cpp)
target_link_libraries(tom_core_tests PRIVATE tom_headers)
add_executable(tom_native_image_tests tests/native_image_tests.cpp)
target_link_libraries(tom_native_image_tests PRIVATE tom_headers)
enable_testing()
add_test(NAME tom_integer_core COMMAND tom_core_tests)
add_test(NAME tom_native_output_integrity COMMAND tom_native_image_tests)
find_package(Python3 COMPONENTS Interpreter QUIET)
if(Python3_Interpreter_FOUND)
  add_test(NAME tom_compiler COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/compiler_tests.py)
  add_test(NAME tom_native_semantics COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/native_semantics_tests.py --exe $<TARGET_FILE:tom_native_cpu>)
  add_test(NAME tom_native_resume COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/native_resume_tests.py --exe $<TARGET_FILE:tom_native_cpu>)
  add_test(NAME tom_sdf_core COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/sdf_core_tests.py)
  add_test(NAME tom_sdf_lowering COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/sdf_lowering_tests.py)
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../../TOM_V0_7_Formalization.pdf" AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../../TOM_V0_7_Package.zip")
    add_test(NAME tom_bounded_space COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/bounded_space_tests.py --exe $<TARGET_FILE:tom_cpu>)
  endif()
endif()
if(TOM_ENABLE_CUDA)
  set(CMAKE_CUDA_ARCHITECTURES ${TOM_CUDA_ARCH})
  enable_language(CUDA)
  find_package(CUDAToolkit REQUIRED)
  if(CMAKE_CUDA_COMPILER_VERSION VERSION_LESS 12.8 AND TOM_CUDA_ARCH STREQUAL "120")
    message(FATAL_ERROR "SM120 needs CUDA 12.8 or a later compatible toolkit")
  endif()
  add_executable(tom_cuda cuda/main.cu)
  target_link_libraries(tom_cuda PRIVATE tom_headers CUDA::cudart)
  set_target_properties(tom_cuda PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
  target_compile_options(tom_cuda PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-lineinfo>)
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/cuda/native_main.cu")
    add_executable(tom_native_cuda cuda/native_main.cu)
    target_link_libraries(tom_native_cuda PRIVATE tom_headers CUDA::cudart)
    set_target_properties(tom_native_cuda PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
    target_compile_options(tom_native_cuda PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-lineinfo>)
  endif()
endif()
